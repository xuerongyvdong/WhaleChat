```bat
set ANDROID_HOME=D:\dp\dsh\_tools\android-sdk
set GRADLE_USER_HOME=D:\dp\dsh\_tools\gradle-home
D:\dp\dsh\_tools\gradle-8.14.2\bin\gradle.bat -p D:\dp\dsh\fishnative assembleDebug --no-daemon
```
